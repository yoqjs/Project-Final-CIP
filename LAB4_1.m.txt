%% COMPUTER IMAGE PROCESSING - FINAL PROJECT
% TOPIC 4: THRESHOLD-BASED & COMPREHENSIVE SEGMENTATION
% ---------------------------------------------------------
clear all; close all; clc;

%% 1. ACQUISITION, GRAYSCALE & INTERPOLATION
img_raw = imread('Brain_Scan.jpg'); 
if size(img_raw, 3) == 3
    img_raw = rgb2gray(img_raw); 
end
% Basic transformation using bilinear interpolation
img = imresize(img_raw, [512, 512], 'bilinear');

%% 2. HISTOGRAM EQUALIZATION & INTENSITY DISTRIBUTIONS
img_eq = histeq(img);

%% 3. GLOBAL AND ADAPTIVE THRESHOLDING
% Manual Global Threshold
T_manual = 120; 
img_manual = img > T_manual;

% Automatic Global Threshold (Otsu's Method)
level_otsu = graythresh(img); 
img_otsu = im2bw(img, level_otsu);

% Adaptive Local Threshold (Gaussian)
T_adaptive = adaptthresh(img, 0.5, 'Statistic', 'gaussian');
img_adaptive = imbinarize(img, T_adaptive);

%% 4. EDGE-BASED SEGMENTATION
img_edges = edge(img, 'Canny');

%% 5. REGION-BASED SEGMENTATION: SPLIT & MERGE
img_double = im2double(img);
% Quadtree decomposition based on local region variance
S = qtdecomp(img_double, 0.18); 
img_split_merge = zeros(size(img_double));

% Simple merge/approximation by assigning the mean intensity to each block
for dim = [512 256 128 64 32 16 8 4 2]
    [vals, r, c] = qtgetblk(img_double, S, dim);
    for k = 1:length(r)
        img_split_merge(r(k):r(k)+dim-1, c(k):c(k)+dim-1) = mean(vals(:,k));
    end
end

%% 6. K-MEANS CLUSTERING (Custom Toolbox-Free Implementation)
X = double(img(:)); 
K = 3; % 3 classes: Background, tissue, and bright structures

% Initialize centroids evenly across the intensity range
centers = linspace(min(X), max(X), K)';
old_centers = zeros(size(centers));
max_iters = 100;
iter = 0;

% K-means optimization loop
while max(abs(centers - old_centers)) > 1e-4 && iter < max_iters
    old_centers = centers;
    % Compute absolute distances to the cluster centers
    distances = abs(X - centers'); 
    % Assign each pixel to the closest center
    [~, idx] = min(distances, [], 2);  
    % Update centers with the mean of the assigned pixels
    for k = 1:K
        if any(idx == k)
            centers(k) = mean(X(idx == k));
        end
    end
    iter = iter + 1;
end
img_kmeans = reshape(idx, size(img));

%% =========================================================================
%% VISUALIZATION AND RESULTS
%% =========================================================================

% FIGURE 1: Intensity, Contrast and Thresholding Analysis
figure('Name', 'Part 1: Distribution, Contrast and Thresholding');
subplot(2,3,1); imshow(img); title('Original Image (Resized)');
subplot(2,3,2); imhist(img); grid on; title('Intensity Histogram');
subplot(2,3,3); imshow(img_eq); title('Histogram Equalized');
subplot(2,3,4); imshow(img_manual); title(['Manual Threshold (T=', num2str(T_manual), ')']);
subplot(2,3,5); imshow(img_otsu); title(['Global Otsu Threshold (T=', num2str(round(level_otsu*255)), ')']);
subplot(2,3,6); imshow(img_adaptive); title('Adaptive Local Threshold');

% FIGURE 2: Advanced Approaches (Edges, Regions and Clusters)
figure('Name', 'Part 2: Edges, Regions and K-means');
subplot(1,3,1); imshow(img_edges); title('Edge-Based (Canny)');
subplot(1,3,2); imshow(img_split_merge, []); title('Region-Based (Split & Merge)');
subplot(1,3,3); imshow(img_kmeans, []); colormap(gca, 'jet'); title('K-means Clustering (K=3)');

%% DATA FOR REPORT TABLE
fprintf('Selected Manual Threshold: %d\n', T_manual);
fprintf('Calculated Optimal Otsu Threshold: %d\n', round(level_otsu*255));
